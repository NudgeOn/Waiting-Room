package main

import (
 "context"
 "encoding/json"
 "fmt"
 "os"
 "strconv"
 "strings"
 "time"
 v "github.com/valkey-io/valkey-go"
 "waiting-room/internal/localcontrol"
)

func main() {
 state, err := localcontrol.LoadState("/state")
 if err != nil { fmt.Println(`{"error":"private_state_unavailable"}`); os.Exit(1) }
 c, err := v.NewClient(v.ClientOption{InitAddress: []string{"valkey:6379"}, Username:"wr_initializer", Password:localcontrol.QueueOwnerPassword(state), DisableCache:true, DisableRetry:true, ForceSingleClient:true})
 if err != nil { fmt.Println(`{"error":"connection_unavailable"}`); os.Exit(1) }
 defer c.Close()
 allowed:=map[string]bool{}
 for _, key:=range strings.Fields("aof_enabled aof_rewrite_in_progress aof_last_write_status aof_last_bgrewrite_status aof_delayed_fsync aof_buffer_length aof_current_size aof_base_size aof_pending_bio_fsync rdb_bgsave_in_progress rdb_last_bgsave_status latest_fork_usec instantaneous_ops_per_sec total_commands_processed total_error_replies rejected_connections sync_full used_memory used_memory_peak used_memory_rss maxmemory connected_clients blocked_clients evicted_keys expired_keys io_threaded_reads_processed io_threaded_writes_processed") { allowed[key]=true }
 row:=map[string]any{"observedAt":time.Now().UTC().Format(time.RFC3339Nano)}
 start:=time.Now(); ctx,cancel:=context.WithTimeout(context.Background(),2*time.Second); defer cancel()
 raw,err:=c.Do(ctx,c.B().Info().Build()).ToString()
 row["infoRoundTripMs"]=float64(time.Since(start).Microseconds())/1000
 if err!=nil { row["error"]="info_unavailable" } else {
  for _,line:=range strings.Split(raw,"\n") { key,value,ok:=strings.Cut(strings.TrimSpace(line),":"); if !ok||!allowed[key] {continue}; if n,e:=strconv.ParseInt(value,10,64); e==nil {row[key]=n} else if value=="ok"||value=="err" {row[key]=value} }
 }
 _=json.NewEncoder(os.Stdout).Encode(row)
}
