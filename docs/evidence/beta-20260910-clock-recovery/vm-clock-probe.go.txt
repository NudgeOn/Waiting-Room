package main
import("encoding/json";"os";"time")
func main(){start:=time.Now();last:=start;count:=0;report:=time.NewTicker(time.Minute);tick:=time.NewTicker(10*time.Millisecond);out:=json.NewEncoder(os.Stdout);out.Encode(map[string]any{"event":"started","wall":start.UTC()});for{select{case<-tick.C: now:=time.Now();if now.UnixNano()<last.UnixNano(){count++;out.Encode(map[string]any{"event":"wall_rollback","previousWall":last.UTC(),"currentWall":now.UTC(),"wallDeltaNs":now.UnixNano()-last.UnixNano(),"monotonicElapsedNs":now.Sub(last).Nanoseconds(),"count":count})};last=now;case<-report.C:now:=time.Now();out.Encode(map[string]any{"event":"minute","wall":now.UTC(),"elapsedSeconds":now.Sub(start).Seconds(),"wallMinusMonotonicNs":now.UnixNano()-start.UnixNano()-now.Sub(start).Nanoseconds(),"rollbacks":count});if now.Sub(start)>110*time.Minute{return}}}}
